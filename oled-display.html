Connecting and Programming a 0.96" SSD1306 OLED (I2C) with ESP32

Complete wiring schematic, required libraries, and tested Arduino C++ verification code.

Hardware Pinout Connections

Arduino IDE Libraries

Install via Arduino Library Manager:

  • Adafruit SSD1306
  • Adafruit GFX Library

Arduino C++ Verification Code

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_RESET    -1
#define SCREEN_ADDRESS 0x3C

Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RESET);

void setup() {
  Serial.begin(115200);
  Wire.begin(21, 22);

  if (!display.begin(SSD1306_SWITCHCAPVCC, SCREEN_ADDRESS)) {
    Serial.println(F("SSD1306 allocation failed"));
    for (;;);
  }

  display.clearDisplay();
  display.setTextColor(SSD1306_WHITE);

  display.setTextSize(1);
  display.setCursor(0, 4);
  display.println("TechCircuitGuide.com");

  display.drawLine(0, 16, SCREEN_WIDTH, 16, SSD1306_WHITE);

  display.setTextSize(2);
  display.setCursor(10, 26);
  display.println("ESP32 OK");

  display.setTextSize(1);
  display.setCursor(18, 48);
  display.println("I2C OLED Ready");

  display.display();
}

void loop() {}
